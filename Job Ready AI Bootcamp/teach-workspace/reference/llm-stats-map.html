import numpy as np

# Softmax with temperature
def softmax(logits, T=1.0):
    z = np.asarray(logits, float) / T
    e = np.exp(z - z.max())
    return e / e.sum()

# LayerNorm (শেষ axis বরাবর)
def layer_norm(x, gamma, beta, eps=1e-5):
    mu = x.mean(-1, keepdims=True)
    var = x.var(-1, keepdims=True)
    return gamma * (x - mu) / np.sqrt(var + eps) + beta

# RMSNorm — গড় বিয়োগ নেই
def rms_norm(x, gamma, eps=1e-6):
    return gamma * x / np.sqrt((x**2).mean(-1, keepdims=True) + eps)

# Top-p (nucleus) কাটা
def top_p(p, thresh=0.9):
    idx = np.argsort(p)[::-1]
    keep = idx[np.cumsum(p[idx]) <= thresh]
    return keep if len(keep) else idx[:1]
